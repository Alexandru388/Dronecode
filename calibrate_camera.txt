# calibrate_camera.py
# Calculeaza matricea intrinseca si coeficientii de distorsiune ai camerei
# folosind imaginile unui checkerboard.
#
# Configuratia folosita:
#   Imagini: /home/alex/aruco_calibration/images/
#   Rezolutie: 1280x720
#   CHECKERBOARD=(8, 6) reprezinta NUMARUL DE COLTURI INTERIOARE,
#   nu numarul de patrate. Asta corespunde unui checkerboard de 9x7 patrate.
#   Rezultat: /home/alex/aruco_calibration/camera_calibration.yaml

from pathlib import Path
import cv2
import numpy as np

BASE_DIR = Path("/home/alex/aruco_calibration")
IMAGES_DIR = BASE_DIR / "images"
OUTPUT_FILE = BASE_DIR / "camera_calibration.yaml"
PREVIEW_DIR = BASE_DIR / "detected_corners"

IMAGE_WIDTH = 1280
IMAGE_HEIGHT = 720

CHECKERBOARD = (8, 6)  # colturi interioare: 8 pe latime, 6 pe inaltime

# Punctele sunt definite in unitati arbitrare. Pentru matricea intrinseca
# si coeficientii de distorsiune, dimensiunea reala a patratului nu este
# necesara. Ea este necesara daca vrei distante 3D in unitati fizice.
objp = np.zeros((CHECKERBOARD[0] * CHECKERBOARD[1], 3), np.float32)
objp[:, :2] = np.mgrid[
    0:CHECKERBOARD[0],
    0:CHECKERBOARD[1]
].T.reshape(-1, 2)

object_points = []
image_points = []

SUPPORTED_EXTENSIONS = {".jpg", ".jpeg", ".png", ".bmp", ".tif", ".tiff"}


def main():
    if not IMAGES_DIR.exists():
        raise FileNotFoundError(f"Directorul cu imagini nu exista: {IMAGES_DIR}")

    image_files = sorted(
        p for p in IMAGES_DIR.iterdir()
        if p.is_file() and p.suffix.lower() in SUPPORTED_EXTENSIONS
    )

    if not image_files:
        raise RuntimeError(f"Nu am gasit imagini in {IMAGES_DIR}")

    PREVIEW_DIR.mkdir(parents=True, exist_ok=True)

    image_size = None
    usable_images = 0

    # Optiuni pentru detectarea colturilor checkerboard-ului
    criteria = (
        cv2.TERM_CRITERIA_EPS + cv2.TERM_CRITERIA_MAX_ITER,
        30,
        0.001
    )

    for image_path in image_files:
        image = cv2.imread(str(image_path))

        if image is None:
            print(f"[SKIP] Nu pot citi: {image_path.name}")
            continue

        gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)

        if image_size is None:
            image_size = (gray.shape[1], gray.shape[0])

        if (gray.shape[1], gray.shape[0]) != (IMAGE_WIDTH, IMAGE_HEIGHT):
            print(
                f"[SKIP] {image_path.name}: rezolutie "
                f"{gray.shape[1]}x{gray.shape[0]}, asteptat "
                f"{IMAGE_WIDTH}x{IMAGE_HEIGHT}."
            )
            continue

        found, corners = cv2.findChessboardCorners(
            gray,
            CHECKERBOARD,
            cv2.CALIB_CB_ADAPTIVE_THRESH
            + cv2.CALIB_CB_NORMALIZE_IMAGE
        )

        if not found:
            print(f"[FAIL] Colturi nedetectate: {image_path.name}")
            continue

        refined_corners = cv2.cornerSubPix(
            gray,
            corners,
            (11, 11),
            (-1, -1),
            criteria
        )

        object_points.append(objp.copy())
        image_points.append(refined_corners)
        usable_images += 1

        preview = image.copy()
        cv2.drawChessboardCorners(preview, CHECKERBOARD, refined_corners, found)
        preview_path = PREVIEW_DIR / f"{image_path.stem}_corners.jpg"
        cv2.imwrite(str(preview_path), preview)

        print(f"[OK] Colturi detectate: {image_path.name}")

    if usable_images < 5:
        raise RuntimeError(
            f"Prea putine imagini utilizabile ({usable_images}). "
            "Foloseste mai multe imagini clare, cu checkerboard-ul "
            "in pozitii si unghiuri diferite."
        )

    print(f"\nImagini analizate: {len(image_files)}")
    print(f"Imagini utilizabile: {usable_images}")
    print("Se calculeaza calibrarea...")

    rms, camera_matrix, dist_coeffs, rvecs, tvecs = cv2.calibrateCamera(
        object_points,
        image_points,
        (IMAGE_WIDTH, IMAGE_HEIGHT),
        None,
        None
    )

    fs = cv2.FileStorage(str(OUTPUT_FILE), cv2.FILE_STORAGE_WRITE)
    if not fs.isOpened():
        raise RuntimeError(f"Nu pot scrie fisierul: {OUTPUT_FILE}")

    fs.write("camera_matrix", camera_matrix)
    fs.write("dist_coeffs", dist_coeffs)
    fs.write("image_width", IMAGE_WIDTH)
    fs.write("image_height", IMAGE_HEIGHT)
    fs.write("rms_reprojection_error", float(rms))
    fs.release()

    print("\nCalibrare terminata.")
    print(f"RMS reprojection error: {rms:.6f} pixeli")
    print("\nCamera matrix:")
    print(camera_matrix)
    print("\nDistortion coefficients:")
    print(dist_coeffs)
    print(f"\nFisier salvat: {OUTPUT_FILE}")
    print(f"Imagini cu colturi detectate: {PREVIEW_DIR}")


if __name__ == "__main__":
    main()
